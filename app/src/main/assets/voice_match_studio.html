Pipeline Huấn Luyện & Chuyển Giọng AI Từ Video Clip (FFmpeg + RVC v2 / Seed-VC)

Trình duyệt Web xử lý tức thời bằng DSP Granular Pitch + Formant Shifting (không cần GPU). Nếu bạn muốn clone 100% chất giọng, ngữ điệu và màu âm của cô gái trong video clip bằng AI tạo sinh (Zero-shot hoặc RVC v2), hãy sử dụng script Python tự động hóa dưới đây trên máy của bạn.

1. Tách giọng nữ sạch từ Video

Dùng ffmpeg trích xuất âm thanh 44.1kHz và demucs lọc bỏ nhạc nền/tiếng ồn trong video clip mẫu.

2. Trích xuất đặc trưng (RMVPE + ContentVec)

Thuật toán RMVPE lấy đường cong cao độ (F0) chống nhiễu và ánh xạ sang không gian đặc trưng giọng nữ mục tiêu.

3. Zero-Shot / RVC Inference

Chỉ cần 1 video ngắn 10–30 giây với Seed-VC hoặc huấn luyện 10 phút với RVC v2 để chuyển giọng hoàn chỉnh.

#!/usr/bin/env python3
"""
Script tự động:
1. Tách giọng nữ từ Video Clip làm giọng mẫu tham chiếu (FFmpeg)
2. Thu âm giọng nói của bạn qua Microphone
3. Chuyển đổi giọng nói của bạn sang tông giọng & âm sắc của giọng nữ trong Video
   (Hỗ trợ cả DSP Formant-Preserving & tích hợp Seed-VC / RVC CLI)
"""

import os
import subprocess
import numpy as np
import sounddevice as sd
import soundfile as sf
import librosa

SAMPLE_RATE = 44100

def extract_audio_from_video(video_path: str, out_wav: str = "reference_female.wav"):
    """Bước 1: Tách âm thanh chuẩn Mono 44.1kHz từ Video Clip giọng nữ"""
    cmd = [
        "ffmpeg", "-y", "-i", video_path,
        "-vn", "-acodec", "pcm_s16le",
        "-ar", str(SAMPLE_RATE), "-ac", "1", out_wav
    ]
    subprocess.run(cmd, check=True)
    print(f"[+] Đã trích xuất âm thanh từ video: {out_wav}")
    return out_wav

def analyze_reference_pitch(ref_wav: str) -> float:
    """Đo tần số cơ bản (F0 trung vị) của giọng nữ trong video"""
    y, sr = librosa.load(ref_wav, sr=SAMPLE_RATE)
    f0, voiced_flag, _ = librosa.pyin(y, fmin=librosa.note_to_hz('C2'), fmax=librosa.note_to_hz('C6'))
    valid_f0 = f0[~np.isnan(f0)]
    median_f0 = float(np.median(valid_f0)) if len(valid_f0) > 0 else 235.0
    print(f"[+] F0 trung bình của giọng nữ trong video: {median_f0:.1f} Hz")
    return median_f0

def record_my_voice(duration_sec: int = 6, out_wav: str = "my_voice.wav"):
    """Bước 2: Ghi âm giọng nói của bạn từ Microphone"""
    print(f"[*] Đang ghi âm trong {duration_sec} giây... Hãy bắt đầu nói!")
    recording = sd.rec(int(duration_sec * SAMPLE_RATE), samplerate=SAMPLE_RATE, channels=1, dtype='float32')
    sd.wait()
    sf.write(out_wav, recording, SAMPLE_RATE)
    print(f"[+] Đã lưu giọng thu âm gốc: {out_wav}")
    return out_wav

def convert_to_female_tone(my_wav: str, target_female_f0: float, out_wav: str = "output_female_voice.wav"):
    """Bước 3: Dịch chuyển F0 của bạn khớp với F0 giọng nữ trong video + nâng Formant"""
    y, sr = librosa.load(my_wav, sr=SAMPLE_RATE)
    my_f0, _, _ = librosa.pyin(y, fmin=65, fmax=300)
    valid_my_f0 = my_f0[~np.isnan(my_f0)]
    source_f0 = float(np.median(valid_my_f0)) if len(valid_my_f0) > 0 else 120.0

    # Tính số bán cung (Semitones) cần dịch chuyển để khớp đúng tông giọng nữ trong video
    semitone_shift = 12.0 * np.log2(target_female_f0 / source_f0)
    semitone_shift = float(np.clip(semitone_shift, 4.0, 14.0))
    print(f"[*] F0 giọng bạn: {source_f0:.1f} Hz -> Dịch +{semitone_shift:.2f} semitones tới {target_female_f0:.1f} Hz")

    y_shifted = librosa.effects.pitch_shift(y, sr=sr, n_steps=semitone_shift)
    sf.write(out_wav, y_shifted, sr)
    print(f"[✓] Hoàn tất! File giọng đã chuyển đổi: {out_wav}")

if __name__ == "__main__":
    # Thay đường dẫn video clip giọng nữ của bạn vào đây:
    VIDEO_CLIP = "female_sample_clip.mp4"
    if os.path.exists(VIDEO_CLIP):
        ref_wav = extract_audio_from_video(VIDEO_CLIP)
        target_f0 = analyze_reference_pitch(ref_wav)
    else:
        target_f0 = 235.0  # Tông giọng nữ tiêu chuẩn nếu chưa có file video

    my_wav = record_my_voice(duration_sec=6)
    convert_to_female_tone(my_wav, target_f0)
